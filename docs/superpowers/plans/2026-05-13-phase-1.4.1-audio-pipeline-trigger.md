# Phase 1.4.1 — Audio Pipeline Trigger — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Wire the existing voice + mux backend (Phase 1.3.5 F32 design) into the runtime so Russian-dialogue scenes ship with audio. Auto-chain from `pollMediaJobsAction` after video success → `voice` → `final_clip`, with single auto-retry, surfaced UI affordances (cost hint, final-clip player priority, retry button, finalize confirm dialog), and two new Director chat tools (`regen_scene_voice`, `compose_scene_final_clip`).

**Architecture:** Reactive chain inside the existing poll-loop — no new webhooks, no new buttons. A pure `planNextChainStep(scene, dialogue)` function in `@mango/core` decides the next step after each succeeded job; thin server helpers (`submitVoiceJob`, `submitFinalClipJob`) are shared between the existing one-shot actions, the poll-loop, the Director tools, and a new `retrySceneAudioAction`. Per-job `retry_count + delayed_until` columns gate one auto-retry with 15 s backoff. UI components consume already-populated `media_jobs` rows + script versions and add new badges (🎵 ready, 🔇 silent fallback) plus a `<FinalizeConfirmDialog>` for the premature-finalize case. `has_full_audio: boolean` is computed in `generateMasterClipAction` and copied onto `master_clip_versions[]` in the poll-loop, surfaced as a badge in Stage 05.

**Tech Stack:** Next.js 16, TypeScript, Supabase (Postgres + RLS + Storage + Realtime), `@fal-ai/client` (queue API: ElevenLabs multilingual-v2, `fal-ai/ffmpeg-api`), Zod, Vitest, Biome.

```ts
import { resolveAudioMode, type SceneForAudio, type VideoModelMetaForAudio } from './audio-mode';

/** Total cost hint shown alongside the video cost on the action tile. */
export const AUDIO_CHAIN_COST_HINT_USD = 0.06;

/** Minimal scene shape the planner needs. Mirrors the jsonb subset. */
export type ChainScene = {
  scene_id: string;
  audio_mode?: 'native' | 'silent_tts' | 'auto';
  video_versions: { version_id: string }[];
  video_active_version_id: string | null;
  voice_audio_versions: { version_id: string }[];
  voice_audio_active_version_id: string | null;
  final_clip: {
    composed_from: {
      video_version_id: string;
      voice_audio_version_id: string | null;
    };
  } | null;
  model_meta: VideoModelMetaForAudio;
};

export type ChainStep =
  | null
  | { kind: 'voice' }
  | {
      kind: 'final_clip';
      video_version_id: string;
      voice_audio_version_id: string | null;
    };

type Dialogue = { speaker: string; text: string } | null;

/**
 * Decides the next reactive step in the per-scene audio pipeline.
 *
 * Called from `pollMediaJobsAction` after a video or voice job lands. Pure —
 * no I/O. Returns `null` when there is nothing to do (no dialogue, no video
 * yet, chain already complete, or the user opted out via `audio_mode='native'`
 * but the native passthrough already exists).
 */
export function planNextChainStep(scene: ChainScene, dialogue: Dialogue): ChainStep {
  // No dialogue → nothing to voice.
  if (!dialogue || !dialogue.text || !dialogue.text.trim()) return null;

  const videoVersionId = scene.video_active_version_id;
  if (!videoVersionId) return null;

  const sceneForAudio: SceneForAudio = {
    audio_mode: scene.audio_mode ?? 'auto',
    dialogue,
  };
  const mode = resolveAudioMode(sceneForAudio, scene.model_meta);

  if (mode === 'native') {
    // Native scenes still get a final_clip row (passthrough) so the master
    // readiness check is uniform. No fal job required; the helper writes
    // the script directly.
    if (
      scene.final_clip &&
      scene.final_clip.composed_from.video_version_id === videoVersionId &&
      scene.final_clip.composed_from.voice_audio_version_id === null
    ) {
      return null;
    }
    return {
      kind: 'final_clip',
      video_version_id: videoVersionId,
      voice_audio_version_id: null,
    };
  }

  // silent_tts: voice first, then mux.
  const voiceVersionId = scene.voice_audio_active_version_id;
  if (!voiceVersionId) return { kind: 'voice' };

  if (
    scene.final_clip &&
    scene.final_clip.composed_from.video_version_id === videoVersionId &&
    scene.final_clip.composed_from.voice_audio_version_id === voiceVersionId
  ) {
    return null;
  }
  return {
    kind: 'final_clip',
    video_version_id: videoVersionId,
    voice_audio_version_id: voiceVersionId,
  };
}
```

```ts
import { getMediaProvider } from '@/server/lib/media-provider-factory';
import { recordPendingJob } from '@/server/lib/scene-helpers';
import {
  type Character,
  type Tier,
  getDefaultVideoModel,
  getDefaultVoiceModel,
  getVideoModelMeta,
  resolveAudioMode,
  resolveVoiceId,
  resolveVoiceSettings,
} from '@mango/core';

type Dialogue = { speaker: string; text: string };

type NarratorVoice = {
  tts_voice_id: string;
  description?: string;
  stability?: number;
  similarity_boost?: number;
  style?: number;
  speed?: number;
} | null;

export type SubmitVoiceJobInput = {
  user_id: string;
  project_id: string;
  scene_id: string;
  dialogue: Dialogue;
  characters: Character[];
  narrator_voice: NarratorVoice;
  effective_tier: Tier;
  /** Model id from the active video version (or scene override) used to resolve audio mode. */
  video_model_id?: string;
  /** Override the resolved voice_id (Director override / retry with custom). */
  voice_id_override?: string;
  /** Override the dialogue text (Director "озвучь сцену с другим текстом"). */
  text_override?: string;
  /** Initial retry_count for the new row. 0 for first try, 1 for first auto-retry. */
  initial_retry_count: number;
  /** Optional delayed_until (ISO string) for backoff after a transient failure. */
  delayed_until?: string;
};

export type SubmitVoiceJobResult =
  | { ok: true; job_id: string; existing: boolean }
  | { ok: false; error: string };

export async function submitVoiceJob(input: SubmitVoiceJobInput): Promise<SubmitVoiceJobResult> {
  const text = (input.text_override ?? input.dialogue.text).trim();
  if (!text) return { ok: false, error: 'scene has no dialogue' };

  const videoModelId = input.video_model_id ?? getDefaultVideoModel(input.effective_tier);
  const videoMeta = getVideoModelMeta(videoModelId);
  const audioMode = resolveAudioMode(
    { audio_mode: 'auto', dialogue: { ...input.dialogue, text } },
    { has_native_audio: videoMeta?.has_native_audio ?? false },
  );
  // Note: even if the user has audio_mode=native, callers that explicitly invoke
  // submitVoiceJob (e.g. Director regen) are requesting TTS — we honour the request.
  // Only the auto-chain consults resolveAudioMode via planNextChainStep.

  const voiceId =
    input.voice_id_override ??
    resolveVoiceId(input.dialogue.speaker, input.characters, input.narrator_voice);

  const voiceSettings = resolveVoiceSettings(
```

```ts
import type { StoredAsset } from '@mango/core';
import { getServerSupabase } from '@mango/db/server';

export type SubmitFinalClipJobInput = {
  user_id: string;
  project_id: string;
  scene_id: string;
  video_version: {
    version_id: string;
    storage: StoredAsset;
    has_native_audio?: boolean;
  };
  /** null → native passthrough (write final_clip = video directly, no fal job). */
  voice_version: {
    version_id: string;
    storage: StoredAsset;
  } | null;
  initial_retry_count: number;
  delayed_until?: string;
};

export type SubmitFinalClipJobResult =
  | { ok: true; mode: 'mux_job'; job_id: string; existing: boolean }
  | { ok: true; mode: 'native_passthrough' }
  | { ok: false; error: string };

function urlOfStorage(storage: StoredAsset): string {
  if (storage.kind === 'fal_passthrough') return storage.url;
  return `supabase://${storage.path}`;
}

export async function submitFinalClipJob(
  input: SubmitFinalClipJobInput,
): Promise<SubmitFinalClipJobResult> {
  // Native passthrough — final_clip is the active video. Write directly to script.
  if (!input.voice_version) {
    const sb = await getServerSupabase();
    const { data: project, error } = await sb
      .from('projects')
      .select('script')
      .eq('id', input.project_id)
      .single();
    if (error || !project) return { ok: false, error: 'project not found' };

    type SceneShape = Record<string, unknown> & { scene_id: string };
    const script = project.script as { scenes: SceneShape[] };
    const updated = {
      ...script,
      scenes: script.scenes.map((s) =>
        s.scene_id === input.scene_id
          ? {
              ...s,
              final_clip: {
                storage: input.video_version.storage,
                composed_from: {
                  video_version_id: input.video_version.version_id,
                  voice_audio_version_id: null,
                },
              },
            }
          : s,
      ),
    };
    const { error: upErr } = await sb
      .from('projects')
      .update({ script: updated as never })
      .eq('id', input.project_id);
    if (upErr) return { ok: false, error: upErr.message };
    return { ok: true, mode: 'native_passthrough' };
  }

  // Mux job via fal.
  const provider = getMediaProvider();
  const ctx = {
    user_id: input.user_id,
    project_id: input.project_id,
    character_id: '',
  };

  const handle = await provider.submitFinalClipMux(
    {
      video_url: urlOfStorage(input.video_version.storage),
      audio_url: urlOfStorage(input.voice_version.storage),
    },
    ctx,
  );

  const { job_id, existing } = await recordPendingJob({
    user_id: input.user_id,
    project_id: input.project_id,
    scene_id: input.scene_id,
    kind: 'final_clip',
    model: handle.model_used,
    fal_request_id: handle.fal_request_id,
    retry_count: input.initial_retry_count,
    delayed_until: input.delayed_until ?? null,
    request_input: {
      ...(handle.request_input ?? {}),
      video_version_id: input.video_version.version_id,
      voice_audio_version_id: input.voice_version.version_id,
    },
  });

  return { ok: true, mode: 'mux_job', job_id, existing };
}
```

```ts
import { submitFinalClipJob, submitVoiceJob } from '@/server/lib/audio-chain-helpers';
import {
  type Character,
  type Tier,
  getVideoModelMeta,
} from '@mango/core';
import { getServerSupabase } from '@mango/db/server';
import { z } from 'zod';

const InputSchema = z.object({
  project_id: z.string().uuid(),
  scene_id: z.string().min(1),
  kind: z.enum(['voice', 'final_clip']),
});

type Input = z.infer<typeof InputSchema>;

export async function retrySceneAudioAction(
  rawInput: unknown,
): Promise<{ ok: true; job_id?: string } | { ok: false; error: string }> {
  let input: Input;
  try {
    input = InputSchema.parse(rawInput);
  } catch {
    return { ok: false, error: 'invalid input' };
  }

  let user: { id: string };
  try {
    user = await getCurrentUser();
  } catch {
    return { ok: false, error: 'unauthorized' };
  }

  const sb = await getServerSupabase();
  const { data: project, error } = await sb
    .from('projects')
    .select('id, user_id, tier, script')
    .eq('id', input.project_id)
    .single();
  if (error || !project) return { ok: false, error: 'project not found' };
  if (project.user_id !== user.id) return { ok: false, error: 'forbidden' };

  type SceneShape = Record<string, unknown> & {
    scene_id: string;
    audio_mode?: 'native' | 'silent_tts' | 'auto';
    dialogue?: { speaker: string; text: string } | null;
    config_overrides?: { tier?: Tier; model?: string };
    video_versions?: { version_id: string; storage: unknown; model?: string }[];
    video_active_version_id?: string | null;
    voice_audio_versions?: { version_id: string; storage: unknown }[];
    voice_audio_active_version_id?: string | null;
  };
  type ScriptShape = {
    scenes: SceneShape[];
    characters?: Character[];
    narrator_voice?: unknown;
  };
  const script = project.script as unknown as ScriptShape;
  if (!script) return { ok: false, error: 'project has no script' };

  const scene = script.scenes.find((s) => s.scene_id === input.scene_id);
  if (!scene) return { ok: false, error: 'scene not found' };

  const projectTier = (project.tier ?? 'economy') as Tier;
  const effectiveTier = scene.config_overrides?.tier ?? projectTier;

  if (input.kind === 'voice') {
    if (!scene.dialogue?.text) return { ok: false, error: 'scene has no dialogue' };
    const r = await submitVoiceJob({
      user_id: user.id,
      project_id: input.project_id,
      scene_id: input.scene_id,
      dialogue: scene.dialogue,
      characters: script.characters ?? [],
      // biome-ignore lint/suspicious/noExplicitAny: narrator shape varies
      narrator_voice: (script.narrator_voice as any) ?? null,
      effective_tier: effectiveTier,
      video_model_id: scene.config_overrides?.model,
      initial_retry_count: 0,
    });
    return r.ok ? { ok: true, job_id: r.job_id } : { ok: false, error: r.error };
  }

  // final_clip retry
  const activeVideo = scene.video_versions?.find(
    (v) => v.version_id === scene.video_active_version_id,
  );
  if (!activeVideo) return { ok: false, error: 'scene has no active video' };
  const activeVoice = scene.voice_audio_versions?.find(
    (v) => v.version_id === scene.voice_audio_active_version_id,
  );
  const meta = getVideoModelMeta(activeVideo.model ?? scene.config_overrides?.model ?? '');
  const r = await submitFinalClipJob({
    user_id: user.id,
    project_id: input.project_id,
    scene_id: input.scene_id,
    // biome-ignore lint/suspicious/noExplicitAny: storage typing crosses jsonb boundary
    video_version: {
      version_id: activeVideo.version_id,
      storage: activeVideo.storage as any,
      has_native_audio: meta?.has_native_audio ?? false,
    },
    voice_version: activeVoice
      ? {
          version_id: activeVoice.version_id,
          // biome-ignore lint/suspicious/noExplicitAny: see above
          storage: activeVoice.storage as any,
        }
      : null,
    initial_retry_count: 0,
  });
  if (!r.ok) return { ok: false, error: r.error };
  return r.mode === 'native_passthrough' ? { ok: true } : { ok: true, job_id: r.job_id };
}
```

```tsx
'use client';

import { generateMasterClipAction } from '@/server/actions/generateMasterClipAction';
import type { Database } from '@mango/db';
import { useEffect, useState, useTransition } from 'react';
import type { SceneView } from './Stage04Provider';

type MediaJobRow = Database['public']['Tables']['media_jobs']['Row'];

interface Props {
  projectId: string;
  scenes: SceneView[];
  jobs: MediaJobRow[];
  onClose: () => void;
  onMasterStarted: () => void;
}

type SceneAudioState =
  | { state: 'ready' }
  | { state: 'in_flight'; kind: 'voice' | 'final_clip' }
  | { state: 'failed' };

function resolveSceneAudio(
  scene: SceneView,
  jobsForScene: MediaJobRow[],
): SceneAudioState {
  if (scene.final_clip) return { state: 'ready' };
  const inFlight = jobsForScene.find(
    (j) =>
      (j.kind === 'voice' || j.kind === 'final_clip') &&
      ['pending', 'running'].includes(j.status),
  );
  if (inFlight) {
    return { state: 'in_flight', kind: inFlight.kind as 'voice' | 'final_clip' };
  }
  const failed = jobsForScene.find(
    (j) =>
      (j.kind === 'voice' || j.kind === 'final_clip') &&
      j.status === 'failed' &&
      (j.retry_count ?? 0) >= 1,
  );
  if (failed) return { state: 'failed' };
  // No final_clip, no jobs — treat as "not started" → blocked.
  return { state: 'failed' };
}

export function FinalizeConfirmDialog({
  projectId,
  scenes,
  jobs,
  onClose,
  onMasterStarted,
}: Props) {
  const [pending, startTransition] = useTransition();
  const [error, setError] = useState<string | null>(null);

  const states = scenes.map((s) => ({
    scene: s,
    state: resolveSceneAudio(s, jobs.filter((j) => j.scene_id === s.scene_id)),
  }));
  const allReady = states.every((x) => x.state.state === 'ready');
  const anyFailed = states.some((x) => x.state.state === 'failed');

  // Poll auto-runs when allReady. Useful for the "Дождаться" path.
  const [autoWait, setAutoWait] = useState(false);
  useEffect(() => {
    if (autoWait && allReady) {
      startTransition(async () => {
        const r = await generateMasterClipAction({ project_id: projectId });
        if (r.ok) onMasterStarted();
        else setError(r.error ?? 'не удалось запустить финализацию');
      });
      setAutoWait(false);
    }
  }, [allReady, autoWait, projectId, onMasterStarted]);

  const runSilent = () => {
    setError(null);
    startTransition(async () => {
      const r = await generateMasterClipAction({ project_id: projectId });
      if (r.ok) onMasterStarted();
      else setError(r.error ?? 'не удалось запустить финализацию');
    });
  };

  return (
    <div className="modal-backdrop" role="dialog" aria-modal="true">
      <div className="modal-panel">
        <button className="modal-close" type="button" onClick={onClose} aria-label="Закрыть">
          ✕
        </button>
        <h2 className="modal-title">Финализировать ролик</h2>
        <p className="modal-sub">
```
